name: Weekly backtest

on:
  schedule:
    - cron: "0 20 * * 6"   # מוצאי שבת 20:00 UTC
  workflow_dispatch: {}

permissions:
  contents: write

concurrency:
  group: radar
  cancel-in-progress: false

jobs:
  backtest:
    runs-on: ubuntu-latest
    timeout-minutes: 150
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
      - run: pip install -r requirements.txt
      - name: Run backtest + calibration
        run: python backtest.py
      - name: Save results (fetch + rebase + retry)
        run: bash .github/scripts/save.sh "Weekly backtest $(date -u +%Y-%m-%d)" data
