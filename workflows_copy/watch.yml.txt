name: Watchlist push alerts

on:
  schedule:
    # כל 30 דקות בזמן המסחר בוול סטריט (13:10–21:40 UTC, ימים ב'–ו'). GitHub מאחר ומדלג על הרצות מתוזמנות
    # (בפועל רק 1–2 מתוך 9 ריצות שעתיות יצאו לפועל), לכן התדירות כפולה – כל ריצה היא "רשת ביטחון" לקודמת
    - cron: "10,40 13-21 * * 1-5"
  workflow_dispatch: {}

permissions:
  contents: write

concurrency:
  group: watch
  cancel-in-progress: false   # לא מבטלים ריצה באמצע שמירה – הבאה תמתין

jobs:
  watch:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    env:
      TELEGRAM_TOKEN: ${{ secrets.TELEGRAM_TOKEN }}
      TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
      ACCOUNT_SIZE: ${{ vars.ACCOUNT_SIZE }}
      ALPACA_KEY_ID: ${{ secrets.ALPACA_KEY_ID }}
      ALPACA_SECRET_KEY: ${{ secrets.ALPACA_SECRET_KEY }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
      - run: pip install -r requirements.txt
      - name: Check watchlist
        run: python watch_alerts.py
      - name: Paper trading monitor (stops, trailing, fills)
        run: python paper_trade.py --hourly
      - name: Save state (fetch + rebase + retry; on conflict re-run on the fresh data)
        env:
          RERUN: python watch_alerts.py && python paper_trade.py --hourly
        run: bash .github/scripts/save.sh "watch $(date -u +%H:%M)" data/watch_state.json data/watch.json data/paper_book.json data/paper_trades.csv
